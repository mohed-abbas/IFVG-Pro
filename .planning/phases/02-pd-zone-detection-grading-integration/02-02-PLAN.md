2. **Section 10 — add `render_dealing_range()` (creates on rotation, extends between rotations). Object field mutation is allowed inside `=>`:**

```
// Renders 3 dashed lines + labels + optional fills + optional OTE for the current range.
// Drawings are created when handles are na; otherwise right edge/label x are updated.
render_dealing_range(DealingRange r) =>
    if not na(r) and r.is_valid and i_show_pd_lines
        int right_edge = math.min(bar_index + i_extend_bars, bar_index + 400)
        int left_edge = math.max(r.high_bar_idx, r.low_bar_idx)
        float hi = r.high_price
        float lo = r.low_price
        float eq = (hi + lo) / 2.0
        // ── Lines ──
        if na(r.line_high)
            r.line_high := line.new(left_edge, hi, right_edge, hi, color=i_pd_line_color, style=line.style_dashed, width=i_pd_line_width)
        else
            line.set_x2(r.line_high, right_edge)
        if na(r.line_eq)
            r.line_eq := line.new(left_edge, eq, right_edge, eq, color=i_pd_eq_color, style=line.style_dashed, width=i_pd_line_width)
        else
            line.set_x2(r.line_eq, right_edge)
        if na(r.line_low)
            r.line_low := line.new(left_edge, lo, right_edge, lo, color=i_pd_line_color, style=line.style_dashed, width=i_pd_line_width)
        else
            line.set_x2(r.line_low, right_edge)
        // ── Labels (exact D-08 format) ──
        string txt_high = "1 (" + str.tostring(hi, "#.##") + ")"
        string txt_eq   = "0.5 (" + str.tostring(eq, "#.##") + ")"
        string txt_low  = "0 (" + str.tostring(lo, "#.##") + ")"
        if na(r.label_high)
            r.label_high := label.new(right_edge, hi, txt_high, style=label.style_none, textcolor=i_pd_line_color, size=size.small)
        else
            label.set_x(r.label_high, right_edge)
            label.set_text(r.label_high, txt_high)
        if na(r.label_eq)
            r.label_eq := label.new(right_edge, eq, txt_eq, style=label.style_none, textcolor=i_pd_eq_color, size=size.small)
        else
            label.set_x(r.label_eq, right_edge)
            label.set_text(r.label_eq, txt_eq)
        if na(r.label_low)
            r.label_low := label.new(right_edge, lo, txt_low, style=label.style_none, textcolor=i_pd_line_color, size=size.small)
        else
            label.set_x(r.label_low, right_edge)
            label.set_text(r.label_low, txt_low)
        // ── Optional zone fills (D-09, default OFF) ──
        if i_show_pd_fills
            if na(r.fill_premium)
                r.fill_premium := linefill.new(r.line_high, r.line_eq, color.new(i_bearish_color, 92))
            if na(r.fill_discount)
                r.fill_discount := linefill.new(r.line_eq, r.line_low, color.new(i_bullish_color, 92))
        else
            if not na(r.fill_premium)
                linefill.delete(r.fill_premium)
                r.fill_premium := na
            if not na(r.fill_discount)
                linefill.delete(r.fill_discount)
                r.fill_discount := na
        // ── Optional OTE band 62%-79% (D-10, default OFF, visual only) ──
        if i_show_ote
            float range_span = hi - lo
            float ote_low_price = lo + range_span * 0.62
            float ote_high_price = lo + range_span * 0.79
            if na(r.ote_line_top)
                r.ote_line_top := line.new(left_edge, ote_high_price, right_edge, ote_high_price, color=color.new(#AB47BC, 70), style=line.style_dotted)
            else
                line.set_x2(r.ote_line_top, right_edge)
                line.set_y1(r.ote_line_top, ote_high_price)
                line.set_y2(r.ote_line_top, ote_high_price)
            if na(r.ote_line_bot)
                r.ote_line_bot := line.new(left_edge, ote_low_price, right_edge, ote_low_price, color=color.new(#AB47BC, 70), style=line.style_dotted)
            else
                line.set_x2(r.ote_line_bot, right_edge)
                line.set_y1(r.ote_line_bot, ote_low_price)
                line.set_y2(r.ote_line_bot, ote_low_price)
            if na(r.ote_fill)
                r.ote_fill := linefill.new(r.ote_line_top, r.ote_line_bot, color.new(#AB47BC, 92))
        else
            if not na(r.ote_line_top)
                line.delete(r.ote_line_top)
                r.ote_line_top := na
            if not na(r.ote_line_bot)
                line.delete(r.ote_line_bot)
                r.ote_line_bot := na
            if not na(r.ote_fill)
                linefill.delete(r.ote_fill)
                r.ote_fill := na
```
