```
// When false, fall back to chart-TF g_liquidity_array. (D-05)
is_pd_tf_higher() =>
    timeframe.in_seconds(i_pd_timeframe) > timeframe.in_seconds(timeframe.period)

// Computes pd_zone for an IFVG midpoint against the current range.
// Strict 50% equilibrium (D-13). Returns "neutral" when no valid range (D-06).
compute_pd_zone(float price_mid) =>
    string result = "neutral"
    if not na(g_current_range) and g_current_range.is_valid
        float hi = g_current_range.high_price
        float lo = g_current_range.low_price
        if hi > lo
            float eq = (hi + lo) / 2.0
            if price_mid > eq
                result := "premium"
            else if price_mid < eq
                result := "discount"
            else
                result := "equilibrium"
    result

// Integer 0-100 or na when no valid range (DSH-02).
compute_range_percent(float price_mid) =>
    int result = na
    if not na(g_current_range) and g_current_range.is_valid
        float hi = g_current_range.high_price
        float lo = g_current_range.low_price
        float span = hi - lo
        if span > 0
            result := int(math.round((price_mid - lo) / span * 100))
    result
```
